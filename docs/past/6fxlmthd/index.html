题解 UVA12396【Remoteland】

2021-01-22 洛谷原文

UVA12396 Remoteland#

传送门#

翻译#

选任意个在 $1$~$n$ 之间的互不相同数，使得他们的乘积为完全平方数且最大。

多组数据，读入到 $0$ 结束。

思路#

由于答案要是完全平方数，所以它所有质因数的次数都是偶数。

考虑枚举 $n$ 以内的质数，由于答案最大为 $n!$ ，于是这个质数在答案里的次数最高也只有它在 $n!$ 中的次数，为了使答案最大，要取到它最大的偶数次数。

这里给出求一个质数 $p$ 在 $n!$ 中的次数的次数的公式: $$f_p=\sum_{i=1}\left\lfloor\dfrac{n}{p^i}\right\rfloor$$ 因为答案要是完全平方数，当 $f_p$ 为奇数时，只能取到 $f_p-1$ 次。

设 $P$ 为不超过 $n$ 的质数集合，答案就是 $$\prod_{p\in P}p^{2\left\lfloor\frac{f_p}{2}\right\rfloor}$$

代码实现#

我们可以使用线性筛 $O(n)$ 求出集合 $P$，$O(log_pn)$ 求出 $f_p$，其它部分复杂度太玄学，不会分析，代码实现见下：

#include <bits/stdc++.h>
using namespace std;
#define ll long long
const int N=1e7+5;
const int mod=1e9+7;
bool p[N];
int a[N];
int n,r,ans,s;
void shai(int n){
	p[1]=1;
	for(int i=2;i<=n;i++){
		if(!p[i]){
			a[++s]=i;
		}
		for(int j=1;j<=s&&i*a[j]<=n;j++){
			p[i*a[j]]=true;
			if(!(i%a[j])){
				break;
			}
		}
	}
}
inline int qpow(int x,int y){
	int res=1;
	while(y){
		if(y&1) res=1ll*res*x%mod;
		x=1ll*x*x%mod;
		y>>=1;
	}
	return res;
}
inline int read(){
	char ch=getchar();
	int x=0,f=1;
	while(ch<'0'||ch>'9'){if(ch=='-') f=-1;ch=getchar();}
	while(ch>='0'&&ch<='9') x=x*10+(ch^48),ch=getchar();
	return x*f;
}
int main(){
	shai(N-5);
	while(n=read()){
		ll ans=1;
		for(int i=1;i<=s&&a[i]<=n;i++){
			int p=a[i];
			int sum=0;
			for(int j=n/p;j;j/=p){
				sum+=j;
			}
			ans=ans*qpow(p,sum-(sum&1))%mod;
		}
		printf("%lld\n",ans);
	}
    return 0;
}
