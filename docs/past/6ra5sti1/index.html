题解 SP3442 【LASTDIG - The last digit】

2020-09-05 洛谷原文

SP3442 LASTDIG - The last digit

传送门

一道快速幂&取余运算的模板题，类似于P1226。

思路

分治&&递归（快速幂）

快速幂应该都会吧.... 快速幂是一种能迅速把$a^b$求出来的算法，这里我用分治实现。

当 $2|b$ ,$a^b=a^\frac{b}{2}×a^\frac{b}{2}$,不需要重复乘2遍，可以快速运算。

否则,$a^b=a^\frac{b-1}{2}×a^\frac{b-1}{2}×a$。

这样递归下去直到 $b=0$或$1$，如果 $b=0$ ，返回值 $1$，否则，返回 $a$ 。

上AC代码~

#include <bits/stdc++.h>
using namespace std;
#define LL unsigned long long
LL k;
LL f(LL x,LL y){
	if(y==0){//x^0=1
		return 1;
	}
	if(y==1){//x^1=x
		return x;
	}
	LL z=f(x,y/2);
	if(y%2==1){
		return (z%k)*(z%k)%k*(x%k);//x^y=(x^((y-1)/2)^2)*x
	}else{
		return (z%k)*(z%k)%k;//x^y=(x^(y/2))^2
	}
}
int main(){
	LL t;
	scanf("%lld",&t);
	for(int i=1;i<=t;i++){
		LL a,b;
		scanf("%lld %lld",&a,&b);
		k=10;//%10保留最后一位
		printf("%lld\n",f(a,b)%k);
	}
	return 0;
}
