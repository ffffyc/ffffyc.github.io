# CF877D Olya and Energy Drinks
#### [_传送门_](https://www.luogu.com.cn/problem/CF877D)
## 题意简述
给出 $n$、$m$、$k$，一张 $n×m$ 的地图和入口与出口，每次可向一个方向走 $1$ ~ $k$ 步，求出从起点到终点最少需要走几步。
## 思路
~~由标签可知~~由数据范围可知，这道题需要使用广搜。

我们定义三个队列 $x$、$y$、$t$ 用于存 $x$、$y$ 坐标与现在走的步数，首先把入口存入 $x$、$y$ 队列并将 $0$ 存入 $t$ 队列，每次操作从每个队列取出一个数，每个方向循环 $k$ 次或循环到障碍和迷宫边缘，把每个点的 $x$、$y$ 坐标存入 $x$、$y$ 队列，并把目前走的步数 $+1$ 后存入 $t$ 队列，重复执行直到走到终点或队列被清空，分别输出步数或 $-1$ 即可。

为什么每次需要把 $1$ ~ $k$ 步走到的地方都存入队列中而不是直接走 $k$ 步到障碍和迷宫边缘存入队列呢？看起来一次走多步不是走得更快么？我们来看下面一组数据：
```cpp
5 6 5
......
##.#.#
#..#..
...#..
......
1 1 4 3
```
走 $k$ 步运行的结果是 $5$ ，而走 $1$ ~ $k$ 步只需要 $2$ 步。

## 代码
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=1e3+10;
int n,m,k,sx,sy,qx,qy;
char ch[N][N];//存图
bool vis[N][N];//是否走过
int d[4][2]={{0,1},{0,-1},{1,0},{-1,0}};//方向
inline char rd(){
	char ch=getchar();
	while(ch!='#'&&ch!='.')ch=getchar();
	return ch;
}
inline int read(){//整形快读
	char ch=getchar();
	int x=0,f=1;
	while(ch<'0'||ch>'9'){if(ch=='-') f=-1;ch=getchar();}
	while(ch>='0'&&ch<='9') x=x*10+(ch^48),ch=getchar();
	return x*f;
}
inline bool ck(int x,int y){//判断一个点是否在图内
	return x>0&&x<=n&&y>0&&y<=m;
}
inline int bfs(){//广搜
	queue<int>x,y,t;//定义三个队列
	x.push(sx);
	y.push(sy);
	t.push(0);
	while(!x.empty()){
		int wx=x.front(),wy=y.front(),cost=t.front();
		x.pop();
		y.pop();
		t.pop();
		if(wx==qx&&wy==qy){//入口出口在一起时直接找到
			return cost;
		}
		for(int i=0;i<4;i++){
			for(int j=1;j<=k;j++){//走k步
				int dx=wx+d[i][0]*j,dy=wy+d[i][1]*j;//走到的地方
				if(!ck(dx,dy)){//走出图就结束循环
					break;
				}
				if(ch[dx][dy]=='#'){//走到障碍就停下来
					break;
				}
				if(!vis[dx][dy]){
					if(dx==qx&&dy==qy){//找到了
						return cost;
					}
					x.push(dx);//插入队列
					y.push(dy);
					t.push(cost);
					vis[dx][dy]=true;
				}
			}
		}
	}
	return -1;//未找到
}
int main(){
	n=read(),m=read(),k=read();
	for(int i=1;i<=n;i++){
		for(int j=1;j<=m;j++){
			ch[i][j]=rd();//读入地图
			if(ch[i][j]=='#'){
				vis[i][j]=true;//判墙壁
			}
		}
	}
	sx=read();//起点
	sy=read();
	qx=read();//终点
	qy=read();
	printf("%d",bfs());//广搜
	return 0;
}
```
